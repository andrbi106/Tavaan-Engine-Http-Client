## استفاده

```php
use Tavaan\Engine\HttpClient\HttpClient;
use Tavaan\Engine\HttpClient\PendingRequest;

// مستقیم
$client = new HttpClient();
$response = $client->get('https://api.example.com/users');
$response = $client->post('https://api.example.com/users', ['name' => 'Ali']);

// Fluent API
$response = $client->request()
    ->withHeaders(['Authorization' => 'Bearer ' . $token])
    ->withToken($token)
    ->asJson()
    ->acceptJson()
    ->timeout(10)
    ->retry(3, 100)
    ->withQueryParameters(['page' => 1, 'limit' => 20])
    ->post($url, $data);

// Response
$response->status();        // 200
$response->successful();     // true
$response->failed();         // false
$response->body();           // string
$response->json();           // array
$response->json('data.id');  // nested key
$response->header('Content-Type');
$response->throw();         // throws on 4xx/5xx

// Pool (موازی)
$responses = $client->pool(function ($pool) {
    $pool->as('user')->get($url1);
    $pool->as('posts')->get($url2);
});
$responses['user']->json();

// مستقیم با PendingRequest
$req = new PendingRequest();
$req->asForm()->post($url, ['key' => 'value']);
$req->withCookies(['session' => 'abc'])->get($url);
$req->withoutTlsVerification()->get($url);  // برای تست
```
